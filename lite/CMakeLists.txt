cmake_minimum_required(VERSION 3.22)
project(lmtorch_lite LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(LMTORCH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(LMTORCH_JSON_INCLUDE "${LMTORCH_ROOT}/build-lite/include" CACHE PATH "Directory containing pinned nlohmann/json.hpp and generated schema_embedded.hpp (lite/build.sh)")
add_executable(lmtorch-lite host.cpp ../src/safetensors.cpp)
target_include_directories(lmtorch-lite PRIVATE . ../third_party/dslvm ../include "${LMTORCH_JSON_INCLUDE}")
target_compile_options(lmtorch-lite PRIVATE -O2 -ffp-contract=off)
if(EMSCRIPTEN)
 target_compile_options(lmtorch-lite PRIVATE -fwasm-exceptions)
 target_link_options(lmtorch-lite PRIVATE -fwasm-exceptions -lnodefs.js -sMODULARIZE=1 -sEXPORT_NAME=createLmtorchLite -sENVIRONMENT=web,worker,node -sEXPORTED_RUNTIME_METHODS=callMain,FS,ENV,NODEFS -sINVOKE_RUN=0 -sFORCE_FILESYSTEM=1 -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=536870912)
 set_target_properties(lmtorch-lite PROPERTIES SUFFIX .cjs)
else()
 add_executable(lmtorch-lite-core-test ../tests/lite/core.cpp)
 target_include_directories(lmtorch-lite-core-test PRIVATE .)
 enable_testing()
 add_test(NAME lite-core COMMAND lmtorch-lite-core-test)
endif()
